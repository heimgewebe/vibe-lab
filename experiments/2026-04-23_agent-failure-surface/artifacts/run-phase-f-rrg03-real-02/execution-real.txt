import json
import hashlib
import difflib
from pathlib import Path

REPO = Path('.')
ART = REPO / 'experiments/2026-04-23_agent-failure-surface/artifacts/run-phase-f-rrg03-real-02'
FIXTURE = ART / 'fixtures' / 'before.md'

locator = 'Process request'
exact_before = 'Validate request before routing.'
exact_after = 'Process request before routing.\nValidate request before routing.'

def resolve(text: str, needle: str):
    encoded = text.encode('utf-8')
    n = needle.encode('utf-8')
    starts = []
    i = 0
    while True:
        j = encoded.find(n, i)
        if j == -1:
            break
        starts.append(j)
        i = j + 1

    def line_for_pos(pos: int):
        return encoded[:pos].count(b'\n') + 1

    if not starts:
        return {'match_count': 0, 'selected_match_index': None,
                'line_start': None, 'line_end': None,
                'byte_start': None, 'byte_end': None, 'matched_text': None}

    idx = 0
    start = starts[idx]
    end = start + len(n)
    return {'match_count': len(starts), 'selected_match_index': idx,
            'line_start': line_for_pos(start), 'line_end': line_for_pos(max(start, end - 1)),
            'byte_start': start, 'byte_end': end, 'matched_text': needle}

text_before = FIXTURE.read_text(encoding='utf-8')
sha_before = hashlib.sha256(text_before.encode('utf-8')).hexdigest()

c1 = resolve(text_before, locator)
text_after = text_before.replace(exact_before, exact_after, 1)
applied = text_after != text_before
sha_after = hashlib.sha256(text_after.encode('utf-8')).hexdigest()
c3 = resolve(text_after, locator)

diff_text = ''.join(difflib.unified_diff(
    text_before.splitlines(keepends=True),
    text_after.splitlines(keepends=True),
    fromfile='before.md', tofile='before.md (mutated)',
))

if c3['match_count'] == 0:
    classification = 'not_found'
elif c1['selected_match_index'] is None or c3['selected_match_index'] is None:
    classification = 'ambiguous'
else:
    same_site = (c1['line_start'] == c3['line_start']
                 and c1['line_end'] == c3['line_end']
                 and c1['byte_start'] == c3['byte_start']
                 and c1['byte_end'] == c3['byte_end'])
    classification = 'stable' if same_site else 'drifted'
