WORKDIR = Path(tempfile.mkdtemp(prefix='rrg03-real-'))
WORK_FILE = WORKDIR / 'before.md'
shutil.copy2(FIXTURE, WORK_FILE)

locator = 'Validate token'
exact_before = 'Validate token before session creation.'
exact_after = 'Check token before session creation.'

text_before = WORK_FILE.read_text(encoding='utf-8')
sha_before = hashlib.sha256(text_before.encode('utf-8')).hexdigest()


def resolve(text: str, needle: str):
	encoded = text.encode('utf-8')
	n = needle.encode('utf-8')
	starts = []
	i = 0
	while True:
		j = encoded.find(n, i)
		if j == -1:
			break
		starts.append(j)
		i = j + 1

	def line_for_pos(pos: int):
		return encoded[:pos].count(b'\n') + 1

	if not starts:
		return {
			'match_count': 0,
			'selected_match_index': None,
			'line_start': None,
			'line_end': None,
			'byte_start': None,
			'byte_end': None,
			'matched_text': None,
		}

	idx = 0
	start = starts[idx]
	end = start + len(n)
	return {
		'match_count': len(starts),
		'selected_match_index': idx,
		'line_start': line_for_pos(start),
		'line_end': line_for_pos(max(start, end - 1)),
		'byte_start': start,
		'byte_end': end,
		'matched_text': needle,
	}

c1 = resolve(text_before, locator)
text_after = text_before.replace(exact_before, exact_after, 1)
applied = text_after != text_before
WORK_FILE.write_text(text_after, encoding='utf-8')
sha_after = hashlib.sha256(text_after.encode('utf-8')).hexdigest()
c3 = resolve(text_after, locator)
